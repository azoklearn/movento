- SCRUB LOGIC — include this exact working script (run after the video element exists). The page must be tall enough to scroll (all the sections below provide that); the video plays forward as you scroll down and backward as you scroll up, always butter-smooth:

```js
const video = document.querySelector('video');
let current = 0, duration = 0;
video.addEventListener('loadedmetadata', () => { duration = video.duration; });
function loop() {
  requestAnimationFrame(loop);
  const dur = duration || video.duration;
  if (!dur) return;
  const max = document.documentElement.scrollHeight - window.innerHeight;
  const progress = max > 0 ? Math.min(1, Math.max(0, window.scrollY / max)) : 0;
  const target = progress * dur;
  current += (target - current) * 0.06;               // lerp smoothing — very smooth
  if (Math.abs(target - current) < 0.01) current = target;
  if (video.readyState >= 2 && Math.abs(video.currentTime - current) > 0.001)
    video.currentTime = current;
}
requestAnimationFrame(loop);
```

  IMPORTANT for smoothness: `preload="auto"`, never call `video.play()`, keep the 0.06 lerp factor, and only set `currentTime` when `readyState >= 2`.
